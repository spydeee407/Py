EXERCISE 2: INHERITANCE

# Python program to implement inheritance
# using employee salary calculation

# Parent class
class Employee:

  def __init__(self, name, basic_salary):
    self.name = name
    self.basic_salary = basic_salary

  def display_employee(self):
    print("\nEmployee Name :", self.name)
    print("Basic Salary :", self.basic_salary)

# Child class
class Salary(Employee):

  def __init__(self, name, basic_salary, bonus):
    # Calling parent class constructor
    Employee.__init__(self, name, basic_salary)
    self.bonus = bonus

  def calculate_salary(self):
    total_salary = self.basic_salary + self.bonus

    self.display_employee()
    print("Bonus      :", self.bonus)
    print("Total Salary :", total_salary)

# Reading input from keyboard
name = input("Enter employee name: ")
basic_salary = float(input("Enter basic salary: "))
bonus = float(input("Enter bonus amount: "))

# Creating object of child class
s = Salary(name, basic_salary, bonus)

# Calling method
s.calculate_salary()

EXERCISE 5: BINARY SEARCH

# Reading number of elements
n = int(input("Enter number of elements: "))

# Creating empty list
data = []

# Reading elements
print("Enter elements:")
for i in range(n):
  element = int(input())
  data.append(element)

# Sorting the list for Binary Search
data.sort()

print("\nSorted List:")
print(data)

# Reading element to search
search = int(input("\nEnter element to search: "))

# Initializing variables
low = 0
high = n - 1
found = False

# Binary Search process
while low <= high:

  mid = (low + high) // 2

  if data[mid] == search:
     print("\nElement found at position", mid + 1)
     found = True
     break

  elif data[mid] < search:
    low = mid + 1

  else:
    high = mid - 1

# If element not found
if found == False:
   print("\nElement not found")

EXERCISE 6: SORTING (BUBBLE SORT)

# Reading number of elements
n = int(input("Enter number of elements: "))

# Creating empty list
data = []

# Reading elements
print("Enter elements:")
for i in range(n):
  element = int(input())
  data.append(element)

# Displaying original list
print("\nOriginal List:")
print(data)

# Bubble Sort process
for i in range(n):
  for j in range(0, n - i - 1):

    if data[j] > data[j + 1]:
       # Swapping elements
       temp = data[j]
       data[j] = data[j + 1]
       data[j + 1] = temp

# Displaying sorted list
print("\nSorted List:")
print(data)

EXERCISE 7: STACK

# Creating empty stack
stack = []

while True:

  print("\nSTACK OPERATIONS")
  print("1. Push")
  print("2. Pop")
  print("3. Display")
  print("4. Exit")

  choice = int(input("Enter your choice: "))

  # Push operation
  if choice == 1:
     element = input("Enter element to push: ")
     stack.append(element)
     print(element, "inserted into stack")

  # Pop operation
  elif choice == 2:
    if len(stack) == 0:
       print("Stack is empty")
    else:
       removed = stack.pop()
       print(removed, "removed from stack")

  # Display operation
  elif choice == 3:
    if len(stack) == 0:
       print("Stack is empty")
    else:
       print("\nStack Elements:")
       for item in reversed(stack):
         print(item)

  # Exit operation
  elif choice == 4:
    print("Program terminated")
    break

  else:
    print("Invalid choice")

EXERCISE 9: BREADTH FIRST SEARCH (BFS)

# Graph represented using dictionary
graph = {
  'A': ['B', 'C'],
  'B': ['D', 'E'],
  'C': ['F'],
  'D': [],
  'E': ['F'],
  'F': []
}

# BFS function
def bfs(graph, start):

  visited = []
  queue = []

  # Add starting node
  visited.append(start)
  queue.append(start)

  print("\nBFS Traversal:")

  while queue:
   node = queue.pop(0)
   print(node, end=" ")

   # Visit adjacent nodes
   for neighbour in graph[node]:
      if neighbour not in visited:
         visited.append(neighbour)
         queue.append(neighbour)

# Reading starting node
start_node = input("Enter starting node: ")

# Calling BFS function
bfs(graph, start_node)

EXERCISE 10: TREE TRAVERSAL

# Creating tree node
class Node:
  def __init__(self, data):
    self.data = data
    self.left = None
    self.right = None

# Inorder Traversal
def inorder(root):
  if root:
     inorder(root.left)
     print(root.data, end=" ")
     inorder(root.right)

# Preorder Traversal
def preorder(root):
  if root:
     print(root.data, end=" ")
     preorder(root.left)
     preorder(root.right)

# Postorder Traversal
def postorder(root):
  if root:
     postorder(root.left)
     postorder(root.right)
     print(root.data, end=" ")

# Creating tree
root = Node('A')
root.left = Node('B')
root.right = Node('C')
root.left.left = Node('D')
root.left.right = Node('E')

# Displaying traversals
print("\nInorder Traversal:")
inorder(root)

print("\n\nPreorder Traversal:")
preorder(root)

print("\n\nPostorder Traversal:")
postorder(root)
